/* -------- 生成配置 -------- */
#define APP_TITLE      @@APP_TITLE@@
#define COLOR_ACCENT   @@COLOR_ACCENT@@
#define COLOR_BG       @@COLOR_BG@@
#define COLOR_TEXT     @@COLOR_TEXT@@
#define COLOR_CARD     @@COLOR_CARD@@
#define PAGE_SIZE      @@PAGE_SIZE@@
#define BG_IMAGE       @@BG_IMAGE@@
#define KILL_JIYU      @@KILL_JIYU@@

typedef struct { const wchar_t *name; const wchar_t *path; const wchar_t *args; const wchar_t *work; } GameEntry;
static const GameEntry GAMES[] = {
@@GAMES@@
};
#define GAME_COUNT ((int)(sizeof(GAMES)/sizeof(GAMES[0])))

/* ---------------- 运行时 ---------------- */
static int g_page = 0;
static int g_pages = 1;
static int g_hover = -1;
static ULONG_PTR g_gdiplusToken;

#define WIN_W  860
#define WIN_H  600
#define CARD_W 240
#define CARD_H 92
#define GAP_X  24
#define GAP_Y  20
#define TOP_Y  96
#define FOOT_H 64

static void GetAppDir(wchar_t *buf, DWORD cch) {
    GetModuleFileNameW(NULL, buf, cch);
    wchar_t *p = wcsrchr(buf, L'\\');
    if (p) *(p+1) = L'\0';
}

/* ---------- 结束极域进程 ---------- */
static void EnableDebugPrivilege(void) {
    HANDLE hToken = NULL;
    if (!OpenProcessToken(GetCurrentProcess(), TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY, &hToken)) return;
    TOKEN_PRIVILEGES tp; LUID luid;
    if (LookupPrivilegeValueW(NULL, SE_DEBUG_NAME, &luid)) {
        tp.PrivilegeCount = 1; tp.Privileges[0].Luid = luid;
        tp.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
        AdjustTokenPrivileges(hToken, FALSE, &tp, sizeof(tp), NULL, NULL);
    }
    CloseHandle(hToken);
}

static int KillJiYu(void) {
    EnableDebugPrivilege();
    static const wchar_t *names[] = { L"studentmain.exe", L"jyuclient.exe", L"student.exe", NULL };
    int killed = 0;
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snap == INVALID_HANDLE_VALUE) return 0;
    PROCESSENTRY32W pe; pe.dwSize = sizeof(pe);
    if (Process32FirstW(snap, &pe)) {
        do {
            for (int i = 0; names[i]; ++i) {
                if (_wcsicmp(pe.szExeFile, names[i]) == 0) {
                    HANDLE h = OpenProcess(PROCESS_TERMINATE, FALSE, pe.th32ProcessID);
                    if (h) { if (TerminateProcess(h, 0)) killed++; CloseHandle(h); }
                    break;
                }
            }
        } while (Process32NextW(snap, &pe));
    }
    CloseHandle(snap);
    return killed;
}

/* ---------- 启动游戏 ---------- */
static void LaunchGame(int idx) {
    if (idx < 0 || idx >= GAME_COUNT) return;
    wchar_t dir[MAX_PATH], full[MAX_PATH], work[MAX_PATH];
    GetAppDir(dir, MAX_PATH);
    swprintf(full, MAX_PATH, L"%s%s", dir, GAMES[idx].path);
    full[MAX_PATH-1] = 0;

    if (GetFileAttributesW(full) == INVALID_FILE_ATTRIBUTES) {
        MessageBoxW(NULL, L"找不到文件：请确认游戏文件与配置路径一致。", L"启动失败", MB_ICONWARNING|MB_OK);
        return;
    }

    const wchar_t *wd = GAMES[idx].work;
    if (wd) { swprintf(work, MAX_PATH, L"%s%s", dir, wd); work[MAX_PATH-1]=0; }
    else { wcsncpy(work, full, MAX_PATH-1); work[MAX_PATH-1]=0; wchar_t *p=wcsrchr(work,L'\\'); if(p)*(p+1)=0; }

    /* .sh 文件：尝试 bash / wsl */
    const wchar_t *ext = wcsrchr(GAMES[idx].path, L'.');
    if (ext && _wcsicmp(ext, L".sh") == 0) {
        wchar_t cmdline[2048];
        swprintf(cmdline, 2048, L"\"%s\" %s", full, GAMES[idx].args ? GAMES[idx].args : L"");
        SHELLEXECUTEINFOW sei; ZeroMemory(&sei, sizeof(sei));
        sei.cbSize=sizeof(sei); sei.lpVerb=L"open"; sei.lpFile=L"bash.exe";
        sei.lpParameters=cmdline; sei.lpDirectory=work; sei.nShow=SW_SHOWNORMAL;
        if (!ShellExecuteExW(&sei)) {
            ZeroMemory(&sei, sizeof(sei)); sei.cbSize=sizeof(sei);
            sei.lpVerb=L"open"; sei.lpFile=L"wsl.exe"; sei.lpParameters=cmdline;
            sei.lpDirectory=work; sei.nShow=SW_SHOWNORMAL;
            if (!ShellExecuteExW(&sei))
                MessageBoxW(NULL, L"无法启动 .sh：请安装 Git Bash 或 WSL。", L"启动失败", MB_ICONERROR|MB_OK);
        }
        return;
    }

    SHELLEXECUTEINFOW sei; ZeroMemory(&sei, sizeof(sei));
    sei.cbSize=sizeof(sei); sei.lpVerb=L"open"; sei.lpFile=full;
    sei.lpParameters=GAMES[idx].args; sei.lpDirectory=work; sei.nShow=SW_SHOWNORMAL;
    if (!ShellExecuteExW(&sei)) {
        wchar_t m[1024]; swprintf(m, 1024, L"启动失败（错误码 %lu）：\n%s", GetLastError(), full);
        MessageBoxW(NULL, m, L"启动失败", MB_ICONERROR|MB_OK);
    }
}

/* ---------- 布局 ---------- */
static void GetCardRect(int slot, RECT *rc) {
    int cols = (WIN_W - 2*GAP_X + GAP_X) / (CARD_W + GAP_X);
    if (cols < 1) cols = 1;
    int row = slot / cols, col = slot % cols;
    int rowCount = (PAGE_SIZE + cols - 1) / cols;
    int gridH = rowCount * CARD_H + (rowCount-1) * GAP_Y;
    int top = TOP_Y + ((WIN_H - TOP_Y - FOOT_H) - gridH) / 2;
    if (top < TOP_Y) top = TOP_Y;
    int x = GAP_X + col * (CARD_W + GAP_X);
    int y = top + row * (CARD_H + GAP_Y);
    rc->left=x; rc->top=y; rc->right=x+CARD_W; rc->bottom=y+CARD_H;
}

static int SlotAt(int x, int y) {
    int count = (g_page < g_pages-1) ? PAGE_SIZE : (GAME_COUNT - g_page*PAGE_SIZE);
    if (count < 0) count = 0;
    for (int i=0;i<count;++i) {
        RECT rc; GetCardRect(i,&rc);
        if (x>=rc.left&&x<=rc.right&&y>=rc.top&&y<=rc.bottom) return i;
    }
    return -1;
}

/* ---------- 绘制 ---------- */
static void Paint(HWND hwnd) {
    PAINTSTRUCT ps; HDC hdc = BeginPaint(hwnd, &ps);
    Gdiplus::Graphics g(hdc);
    g.SetSmoothingMode(Gdiplus::SmoothingModeAntiAlias);
    g.SetTextRenderingHint(Gdiplus::TextRenderingHintClearTypeGridFit);

    if (BG_IMAGE) {
        wchar_t dir[MAX_PATH], img[MAX_PATH];
        GetAppDir(dir, MAX_PATH);
        swprintf(img, MAX_PATH, L"%s%s", dir, BG_IMAGE); img[MAX_PATH-1]=0;
        Gdiplus::Image bmp(img);
        if (bmp.GetWidth() > 0) g.DrawImage(&bmp, 0, 0, WIN_W, WIN_H);
        else g.Clear(Gdiplus::Color(255,GetRValue(COLOR_BG),GetGValue(COLOR_BG),GetBValue(COLOR_BG)));
        Gdiplus::SolidBrush mask(Gdiplus::Color(120,0,0,0));
        g.FillRectangle(&mask, 0, 0, WIN_W, WIN_H);
    } else {
        g.Clear(Gdiplus::Color(255,GetRValue(COLOR_BG),GetGValue(COLOR_BG),GetBValue(COLOR_BG)));
    }

    Gdiplus::FontFamily fam(L"Microsoft YaHei");
    Gdiplus::Font fTitle(&fam, 26, Gdiplus::FontStyleBold, Gdiplus::UnitPixel);
    Gdiplus::Font fSmall(&fam, 14, Gdiplus::FontStyleRegular, Gdiplus::UnitPixel);
    Gdiplus::Font fCard(&fam, 17, Gdiplus::FontStyleBold, Gdiplus::UnitPixel);
    Gdiplus::SolidBrush brText(Gdiplus::Color(255,GetRValue(COLOR_TEXT),GetGValue(COLOR_TEXT),GetBValue(COLOR_TEXT)));
    g.DrawString(APP_TITLE, -1, &fTitle, Gdiplus::PointF(28,22), &brText);

    wchar_t tip[128];
    swprintf(tip, 128, L"共 %d 个游戏  ·  第 %d / %d 页  ·  点击卡片启动", GAME_COUNT, g_page+1, g_pages);
    Gdiplus::SolidBrush brMuted(Gdiplus::Color(180,GetRValue(COLOR_TEXT),GetGValue(COLOR_TEXT),GetBValue(COLOR_TEXT)));
    g.DrawString(tip, -1, &fSmall, Gdiplus::PointF(30,62), &brMuted);

    int count = (g_page < g_pages-1) ? PAGE_SIZE : (GAME_COUNT - g_page*PAGE_SIZE);
    if (count < 0) count = 0;
    for (int i=0;i<count;++i) {
        RECT rc; GetCardRect(i,&rc);
        bool hot = (i == g_hover);
        int r = 14;
        Gdiplus::GraphicsPath path;
        path.AddArc(rc.left,rc.top,r,r,180,90);
        path.AddArc(rc.right-r,rc.top,r,r,270,90);
        path.AddArc(rc.right-r,rc.bottom-r,r,r,0,90);
        path.AddArc(rc.left,rc.bottom-r,r,r,90,90);
        path.CloseFigure();
        Gdiplus::Color cc = hot
            ? Gdiplus::Color(255,GetRValue(COLOR_ACCENT),GetGValue(COLOR_ACCENT),GetBValue(COLOR_ACCENT))
            : Gdiplus::Color(230,GetRValue(COLOR_CARD),GetGValue(COLOR_CARD),GetBValue(COLOR_CARD));
        Gdiplus::SolidBrush brCard(cc);
        g.FillPath(&brCard, &path);
        Gdiplus::SolidBrush brCt(hot ? Gdiplus::Color(255,255,255,255)
            : Gdiplus::Color(255,GetRValue(COLOR_TEXT),GetGValue(COLOR_TEXT),GetBValue(COLOR_TEXT)));
        Gdiplus::StringFormat sf; sf.SetAlignment(Gdiplus::StringAlignmentCenter);
        sf.SetLineAlignment(Gdiplus::StringAlignmentCenter);
        Gdiplus::RectF rf((float)rc.left,(float)rc.top,(float)(rc.right-rc.left),(float)(rc.bottom-rc.top));
        g.DrawString(GAMES[g_page*PAGE_SIZE+i].name, -1, &fCard, rf, &sf, &brCt);
    }

    wchar_t nav[64]; swprintf(nav,64,L"‹   %d / %d   ›", g_page+1, g_pages);
    Gdiplus::StringFormat sfN; sfN.SetAlignment(Gdiplus::StringAlignmentCenter);
    g.DrawString(nav,-1,&fCard,Gdiplus::RectF((float)(WIN_W/2-90),(float)(WIN_H-FOOT_H+10),180,40),&sfN,&brText);

    EndPaint(hwnd, &ps);
}

static LRESULT CALLBACK WndProc(HWND hwnd, UINT msg, WPARAM wp, LPARAM lp) {
    switch (msg) {
    case WM_PAINT: Paint(hwnd); return 0;
    case WM_ERASEBKGND: return 1;
    case WM_MOUSEMOVE: {
        int h = SlotAt(LOWORD(lp),HIWORD(lp));
        if (h != g_hover) { g_hover = h; InvalidateRect(hwnd,NULL,FALSE); }
        return 0;
    }
    case WM_LBUTTONDOWN: {
        int x = LOWORD(lp), y = HIWORD(lp);
        int slot = SlotAt(x,y);
        if (slot >= 0) { LaunchGame(g_page*PAGE_SIZE + slot); return 0; }
        if (y > WIN_H-FOOT_H && g_pages > 1) {
            if (x < WIN_W/2) { if (g_page>0) g_page--; }
            else { if (g_page<g_pages-1) g_page++; }
            g_hover=-1; InvalidateRect(hwnd,NULL,FALSE);
        }
        return 0;
    }
    case WM_KEYDOWN:
        if (wp==VK_LEFT && g_page>0) { g_page--; InvalidateRect(hwnd,NULL,FALSE); }
        if (wp==VK_RIGHT && g_page<g_pages-1) { g_page++; InvalidateRect(hwnd,NULL,FALSE); }
        return 0;
    case WM_DESTROY: PostQuitMessage(0); return 0;
    }
    return DefWindowProcW(hwnd,msg,wp,lp);
}

int WINAPI wWinMain(HINSTANCE hInst, HINSTANCE hPrev, LPWSTR lp, int show) {
    (void)hPrev; (void)lp;
    Gdiplus::GdiplusStartupInput si;
    GdiplusStartup(&g_gdiplusToken, &si, NULL);

#if KILL_JIYU
    KillJiYu();
#endif

    g_pages = (GAME_COUNT + PAGE_SIZE - 1) / PAGE_SIZE;
    if (g_pages < 1) g_pages = 1;

    WNDCLASSEXW wc; ZeroMemory(&wc,sizeof(wc));
    wc.cbSize=sizeof(wc); wc.lpfnWndProc=WndProc; wc.hInstance=hInst;
    wc.hCursor=LoadCursorW(NULL,IDC_ARROW);
    wc.hbrBackground=(HBRUSH)(COLOR_WINDOW+1);
    wc.lpszClassName=L"JftwGoLauncherGen";
    RegisterClassExW(&wc);

    HWND hwnd = CreateWindowExW(0, L"JftwGoLauncherGen", APP_TITLE,
        WS_OVERLAPPED | WS_CAPTION | WS_SYSMENU | WS_MINIMIZEBOX,
        CW_USEDEFAULT, CW_USEDEFAULT, WIN_W, WIN_H + 36,
        NULL, NULL, hInst, NULL);
    if (!hwnd) { Gdiplus::GdiplusShutdown(g_gdiplusToken); return 1; }

    RECT rc; GetWindowRect(hwnd,&rc);
    int sw=GetSystemMetrics(SM_CXSCREEN), sh=GetSystemMetrics(SM_CYSCREEN);
    SetWindowPos(hwnd,NULL,(sw-(rc.right-rc.left))/2,(sh-(rc.bottom-rc.top))/2,0,0,SWP_NOSIZE|SWP_NOZORDER);

    ShowWindow(hwnd, show);
    UpdateWindow(hwnd);

    MSG msg;
    while (GetMessageW(&msg,NULL,0,0)) { TranslateMessage(&msg); DispatchMessageW(&msg); }
    Gdiplus::GdiplusShutdown(g_gdiplusToken);
    return (int)msg.wParam;
}
